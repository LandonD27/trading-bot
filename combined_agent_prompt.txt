def gh_load(filename):
    r = requests.get(f'https://api.github.com/repos/{GH_REPO}/contents/{filename}', headers=GH_H)

    return {}, None

def gh_save(filename, data, sha):
    content = base64.b64encode(json.dumps(data, indent=2).encode()).decode()
    payload = {'message': f'{filename} {datetime.now().strftime(\"%Y-%m-%d %H:%M\")}', 'content': content}
    if sha: payload['sha'] = sha
    requests.put(f'https://api.github.com/repos/{GH_REPO}/contents/{filename}', headers=GH_H, json=payload)

def get_top_politicians():
    r = requests.get('https://www.capitoltrades.com/politicians?sortBy=traded-volume&sortDirection=desc', headers=SCRAPE_H, timeout=15)
    soup = BeautifulSoup(r.text, 'html.parser')
    pols, seen = [], set()
    for link in soup.select(\"a[href*='/politicians/']\"):
        slug = link['href'].rstrip('/').split('/')[-1]
        if not re.match(r'^[A-Z]\d{6}$', slug) or slug in seen: continue
        seen.add(slug)
        text = link.get_text(separator=' ', strip=True)
        nm = re.match(r'^(.*?)\s*(Republican|Democrat|Independent)', text)
        name = nm.group(1).strip() if nm else slug
        vm = re.search(r'Volume\s*([\d.]+[MK]?)', text, re.IGNORECASE)
        def n(s):
            s = s.strip()
            if s.endswith('M'): return float(s[:-1])*1e6
            if s.endswith('K'): return float(s[:-1])*1e3
            return float(s or 0)
        pols.append({'name': name, 'slug': slug, 'volume': n(vm.group(1)) if vm else 0})
    return sorted(pols, key=lambda x: x['volume'], reverse=True)[:TOP_N]

def get_trades(slug):
    r = requests.get(f'https://www.capitoltrades.com/politicians/{slug}', headers=SCRAPE_H, timeout=15)
    text = re.sub(r'\s+', ' ', BeautifulSoup(r.text, 'html.parser').get_text(separator=' '))
    trades, seen = [], set()
    for tr, pd, td, side in re.findall(r'([A-Z]{1,6}:[A-Z]{2})\s+(\d{1,2} \w+ \d{4})\s+(\d{1,2} \w+ \d{4}).{0,60}?\b(buy|sell)\b', text, re.IGNORECASE):
        ticker = tr.split(':')[0].upper()
        tid = f'{slug}|{ticker}|{td}|{side.lower()}'
        if tid not in seen:
            seen.add(tid)
            trades.append({'id': tid, 'ticker': ticker, 'side': side.lower(), 'trade_date': td, 'pub_date': pd})
    return trades
